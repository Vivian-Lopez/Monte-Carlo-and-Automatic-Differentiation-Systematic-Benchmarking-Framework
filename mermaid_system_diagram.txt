    direction TB

    class workload {
        <<decorator>>
        +name: str
        applies @dataclass to cls
        sets cls.workload_type = name
        registers cls in WORKLOAD_REGISTRY
    }

    class WorkloadConfig {
        <<base class>>
        +workload_type: ClassVar[str]
        +validate()
        +to_dict() dict
        +from_dict(data) WorkloadConfig
        +config_hash() str
    }

    class EuropeanOptionConfig {
        <<@workload("european")>>
        +S0: float = 100.0
        +K: float = 100.0
        +r: float = 0.05
        +sigma: float = 0.20
        +T: float = 1.0
        +option_type: str = "call"
        +N: int = 1
        +M: int = 10000
        +seed: int = 42
    }

    class WORKLOAD_REGISTRY {
        <<dict>>
        "european" → EuropeanOptionConfig
        config_from_dict(data) WorkloadConfig
    }

    class MonteCarloEngine {
        <<abstract>>
        +run(config, ad_mode) Tuple[float, dict|None]
        +supports(workload_type) bool
        +supported_ad_modes() Tuple
    }

    class CPUMonteCarloEngine {
        +supported_ad_modes() = ("none",)
        +run(config, ad_mode) (float, None)
        -_price_european(config) float
    }

    class JAXMonteCarloEngine {
        +supported_ad_modes() = ("none","forward","reverse")
        +run(config, ad_mode) (float, dict|None)
        -_run_european(config, ad_mode)
        -_compute_greeks(kernel, config, ad_mode)
    }

    class CUDAMonteCarloEngine {
        +supported_ad_modes() = ("none","forward")
        +run(config, ad_mode) (float, dict|None)
        -_KERNEL_SOURCE: cuRAND C++ kernel
    }

    class CPPMonteCarloEngine {
        +supported_ad_modes() = ("none",)
        +run(config, ad_mode) (float, None)
        -price_european: pybind11 → OpenMP C++
    }
